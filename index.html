function descargarQR() { if (!asistenteActual) { showToast('⚠️ No hay QR para descargar', 'error'); return; } // Crear un contenedor temporal solo con el QR y los datos const tempDiv = document.createElement('div'); tempDiv.style.position = 'absolute'; tempDiv.style.left = '-9999px'; tempDiv.style.top = '-9999px'; tempDiv.style.background = '#ffffff'; tempDiv.style.padding = '30px'; tempDiv.style.borderRadius = '16px'; tempDiv.style.textAlign = 'center'; tempDiv.style.width = '400px'; document.body.appendChild(tempDiv); // Contenido del QR (sin botones) tempDiv.innerHTML = `

🎫 ENTRADA

LOS HIJOS DE TENCHA
🎯 ID de validación ${asistenteActual.id}
Nombre
${asistenteActual.nombre}
Correo
${asistenteActual.email}
Evento
${asistenteActual.evento}
🎸 Los Hijos de Tencha
`; // Generar QR en el contenedor temporal const qrContainer = tempDiv.querySelector('#qrTemp'); const url = window.location.origin + '/validador.html?id=' + asistenteActual.id; new QRCode(qrContainer, { text: url, width: 250, height: 250, colorDark: '#ff6b00', colorLight: '#ffffff', correctLevel: QRCode.CorrectLevel.H }); // Esperar a que se renderice setTimeout(() => { html2canvas(tempDiv, { scale: 2, backgroundColor: '#ffffff', allowTaint: true, useCORS: true, logging: false }).then(canvas => { const link = document.createElement('a'); link.download = `entrada_${asistenteActual.nombre.replace(/\s/g, '_')}.png`; link.href = canvas.toDataURL('image/png'); link.click(); document.body.removeChild(tempDiv); showToast('✅ QR descargado con éxito', 'success'); }).catch(error => { console.error('Error al descargar QR:', error); document.body.removeChild(tempDiv); showToast('❌ Error al descargar QR', 'error'); }); }, 500); }
